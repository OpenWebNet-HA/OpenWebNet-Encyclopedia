def sha(b):
    return hashlib.sha256(b).hexdigest()


def enc(x):
    return (json.dumps(x, indent=2, sort_keys=True) + '\n').encode()


def exclusive(p, b):
    p.parent.mkdir(parents=True, exist_ok=True)
    with p.open('xb') as f:
        f.write(b)


def build():
    pins = {}

    def read(p):
        p = Path(p)
        f = ROOT / p
        assert not p.is_absolute() and '..' not in p.parts
        assert f.is_file() and not f.is_symlink() and ROOT in f.resolve().parents
        b = f.read_bytes()
        pins[str(p)] = {'path': str(p), 'bytes': len(b), 'sha256': sha(b)}
        return b

    manifest = json.loads(read(BASE / 'analysis/application-regular-files.json'))
    original = BASE / 'analysis/application/home/bticino/bin/bt_taskmanager'
    b = read(original)
    assert len(b) == manifest['home/bticino/bin/bt_taskmanager']['bytes'] == 159048
    assert sha(b) == manifest['home/bticino/bin/bt_taskmanager']['sha256'] == 'ad68f9bf43f7f10af122f34d3cb14b7d8c02730c46225a9a6cd1bf3e663072cb'
    assert b[:8] == bytes.fromhex('7f454c4601010161')
    h = struct.unpack_from('<HHIIIIIHHHHHH', b, 16)
    assert h[0:2] == (2, 40) and h[6] == 0x202 and h[8] == 32
    ph = [struct.unpack_from('<8I', b, h[4] + i * h[8]) for i in range(h[9])]
    loads = [{'offset': s[1], 'VA': s[2], 'file_bytes': s[4], 'memory_bytes': s[5], 'flags': s[6]} for s in ph if s[0] == 1]

    def off(a, n=4):
        xs = [s['offset'] + a - s['VA'] for s in loads if s['VA'] <= a and a + n <= s['VA'] + s['file_bytes']]
        assert len(xs) == 1, hex(a)
        return xs[0]

    def word(a):
        return struct.unpack_from('<I', b, off(a))[0]

    def branch(a):
        v = word(a)
        assert v & 0xff000000 == 0xeb000000
        n = v & 0xffffff
        if n & (1 << 23):
            n -= 1 << 24
        return a + 8 + n * 4

    for a, v in GUARDS.items():
        assert word(a) == v, hex(a)
    calls = {0xa938: 0x9d04, 0xa94c: 0x9e24, 0xa96c: 0x9d04, 0xa984: 0x9fa4}
    for a, t in calls.items():
        assert branch(a) == t
    joins = []
    for a, (p, v) in POOLS.items():
        x = word(a)
        assert x & 0x0f7f0000 == 0x051f0000
        assert a + 8 + (x & 4095) * (1 if x & (1 << 23) else -1) == p and word(p) == v
        joins.append({'instruction_VA': hex(a), 'pool_VA': hex(p), 'pool_word': hex(v)})
    literals = []
    for a, text in LITERALS:
        raw = text.encode('ascii') + b'\0'
        assert b[off(a, len(raw)):off(a, len(raw)) + len(raw)] == raw
        literals.append({'VA': hex(a), 'file_offset': hex(off(a, len(raw))), 'text': text, 'NUL_bytes_hex': raw.hex()})
    rel = subprocess.check_output(['readelf', '-rW', str(ROOT / original)], text=True)

    def immediate(v):
        n = v & 255
        r = ((v >> 8) & 15) * 2
        return ((n >> r) | (n << (32 - r))) & 0xffffffff if r else n

    imports = []
    for name, a, slot in [('sprintf', 0x9d04, 0x36620), ('system', 0x9e24, 0x36680), ('fopen', 0x9fa4, 0x36700)]:
        u, v, z = [word(a + i) for i in (0, 4, 8)]
        assert u & 0xfffff000 == 0xe28fc000 and v & 0xfffff000 == 0xe28cc000 and z & 0xfffff000 == 0xe5bcf000
        assert a + 8 + immediate(u) + immediate(v) + (z & 4095) == slot
        rows = [x.strip() for x in rel.splitlines() if 'R_ARM_JUMP_SLOT' in x and re.search(r'\b' + name + r'@GLIBC_', x)]
        assert len(rows) == 1 and int(rows[0].split()[0], 16) == slot
        imports.append({'name': name, 'PLT_VA': hex(a), 'GOT_slot': hex(slot), 'words': [hex(u), hex(v), hex(z)], 'original_relocation': rows[0]})
    spans = []
    for label, a, z in SPANS:
        o = off(a, z - a)
        raw = b[o:o + z - a]
        asm = subprocess.check_output(['objdump', '-d', '-m', 'arm', f'--start-address={a}', f'--stop-address={z}', str(ROOT / original)], text=True)
        spans.append({'label': label, 'start_VA': hex(a), 'end_exclusive_VA': hex(z), 'file_offset': hex(o), 'bytes_hex': raw.hex(), 'sha256': sha(raw), 'host_disassembly': asm})
    prior = BASE / 'research/2026-11-01-first-exploration'
    leads = json.loads(read(prior / 'DATA_FILE_LEADS.json'))
    lead = [x for x in leads if x['product'] == 'MH200N'][0]['literal_leads']
    assert any(x['literal'] == '*.*.spt' and x['sha256'] == sha(b) and x['file_offset'] == off(0x2b8a8) for x in lead)
    read(prior / 'REPORT.md')
    source = {'component': 'Separate MH200N application bundle, NOT an installed overlay', 'original': pins[str(original)],
              'header52_hex': b[:52].hex(), 'raw_flags': hex(h[6]), 'PT_LOAD': loads,

    report = '''# MH200N 1.1.8 night07 — application pattern and listing-path request contract

## Fixed pattern arguments to same-buffer request

Before sprintf0xa938, R0 is that local buffer and R1 is exact compiled format `%s %s%s > %s%s`. R2 supplies `ls -1` and R3 supplies `/root/script/`. def main():

    ap.add_argument('--check', action='store_true')
    a = ap.parse_args()
    names = ['SOURCE.json', 'ARGUMENT_FLOW.json', 'INPUTS.json', 'REPORT.md', 'OUTPUT_MANIFEST.json']
    if not a.check:
        assert not any((HERE / n).exists() for n in names) and not (HERE / 'first-candidate').exists()
    out = build()
    if a.check:
        for n, b in out.items():
            assert (HERE / n).read_bytes() == b, n
        for x in json.loads((HERE / 'first-candidate/INDEX.json').read_bytes())['files']:
            b = (HERE / 'first-candidate' / x['copy']).read_bytes()
            assert len(b) == x['bytes'] and sha(b) == x['sha256'] and b == (HERE / x['copy']).read_bytes()
    else:
        for n, b in out.items():
            exclusive(HERE / n, b)
        copies = {**out, 'collect_application_v1.py': Path(__file__).read_bytes()}
        ix = {'scope': 'Immediate exact first outputs and collector; precomputed index excludes itself', 'files': [{'copy': n, 'bytes': len(b), 'sha256': sha(b)} for n, b in sorted(copies.items())]}
        for n, b in copies.items():
            exclusive(HERE / 'first-candidate' / n, b)
        exclusive(HERE / 'first-candidate/INDEX.json', enc(ix))
    print(json.dumps({'status': 'PASS', 'outputs': len(out), 'spans': len(SPANS), 'guards': len(GUARDS), 'original_execution': 0, 'new_coverage': 'own application pattern/stack/samebuffer request contract; outercaller open'}))
