 def scan(data,source_id):
  found=[]
  for name,sig in SIGS.items():
   start=0
   while True:
    off=data.find(sig,start)
    if off<0:break
    start=off+1;r={'format_lead':name,'offset':off,'signature_hex':sig.hex(),'available_bytes_after_signature_start':len(data)-off}
    if name=='SQLite3':
     h=data[off:off+100];reasons=[]
     if len(h)!=100:reasons.append('truncated_100_byte_header')
     else:
      rawps=int.from_bytes(h[16:18],'big');ps=65536 if rawps==1 else rawps
      validps=512<=ps<=65536 and ps&(ps-1)==0
      if not validps:reasons.append('invalid_page_size')
      if h[18] not in (1,2):reasons.append('read_write_version_outside_selected_ordinary_1_2_admission')
      if h[19] not in (1,2):reasons.append('read_version_outside_selected_ordinary_1_2_admission')
      if h[21:24]!=bytes([64,32,32]):reasons.append('invalid_fixed_payload_fractions')
      if validps and ps-h[20]<480:reasons.append('invalid_usable_page_size')
      if validps and len(data)-off<ps:reasons.append('no_complete_first_page_extent')
      if h[72:92]!=bytes(20):reasons.append('nonzero_reserved_header_expansion')
      if len(data)-off>100 and data[off+100] not in (2,5,10,13):reasons.append('unadmitted_first_btree_page_type')
      count=int.from_bytes(h[28:32],'big');counter=int.from_bytes(h[24:28],'big');vv=int.from_bytes(h[92:96],'big');validcount=count!=0 and counter==vv
      r['header_admission']={'header100_sha256':H(h),'page_size_raw':rawps,'page_size_interpreted':ps,'write_version':h[18],'read_version':h[19],'reserved_bytes_per_page':h[20],'payload_fraction_bytes_hex':h[21:24].hex(),'page_count_header':count,'page_count_current_under_documented_counter_rule':validcount,'declared_extent_if_current':count*ps if validps and validcount else None,'remaining_extent_bytes':len(data)-off,'page_count_not_used_to_reject_candidate':True,'full_header_schema_rows_not_parsed':True}
     r['rejection_reasons']=reasons;r['status']='REJECTED_SELECTED_SQLITE3_HEADER_ADMISSION' if reasons else 'PLAUSIBLE_SQLITE3_HEADER_REQUIRES_FULL_FORMAT_REVIEW'
    else:r['status']='UNPARSED_SIGNATURE_ONLY_NOT_RECOVERED_DATABASE';r['scope']='No alternate-format parser/type/extent or database assignment; embedded magic alone is insufficient'
    found.append(r)
  return {'content_sha256':H(data),'bytes':len(data),'candidate_hits':found}
 def zipscan(data,source_id):
  nonlocal plain_total
  if not(data.startswith(b'PK\x03\x04') or data.startswith(b'PK\x05\x06')):return
  with zipfile.ZipFile(io.BytesIO(data)) as z:
   infos=z.infolist();assert len(infos)<=LIMIT['zip_members'];members=[]
   for n,i in enumerate(infos):
    r={'index':n,'name':i.filename,'bytes':i.file_size,'compressed_bytes':i.compress_size,'CRC32':f'{i.CRC:08x}','encrypted':bool(i.flag_bits&1)}
    if i.is_dir():r['payload_status']='DIRECTORY_METADATA_ONLY'
    elif r['encrypted']:r['payload_status']='ENCRYPTED_NOT_OPENED'
    elif i.file_size>LIMIT['plain_member_bytes'] or plain_total+i.file_size>LIMIT['plain_total_bytes']:r['payload_status']='CAP_NOT_OPENED'
    else:
     with z.open(i) as f:payload=f.read(LIMIT['plain_member_bytes']+1)
     assert len(payload)==i.file_size and len(payload)<=LIMIT['plain_member_bytes'];plain_total+=len(payload)
     # zipfile verifies its CRC on EOF. No output file or archive path extraction occurs.
     r['payload_status']='UNENCRYPTED_CRC_VERIFIED_STATIC_SIGNATURE_SCAN';r['payload_sha256']=H(payload)
     result=scan(payload,source_id+'::zip-member-'+str(n));result['source_id']=source_id+'::zip-member-'+str(n);result['provenance']={'archive_source_id':source_id,'member_index':n,'member_name':i.filename,'CRC32':r['CRC32']};candidates.append(result)
     r['nested_archive_body_unexpanded']=payload.startswith(b'PK\x03\x04')
    members.append(r)
  archives.append({'source_id':source_id,'sha256':H(data),'bytes':len(data),'central_directory_and_selected_plain_members':members})
 files_checked=0;bytes_checked=0
 selected=[p for p in a['packages'] if p['product'] in ('F450','F459')];assert len(selected)==3

    f=root/rel;assert f.is_file() and not f.is_symlink();data=f.read_bytes();digest=H(data);assert (len(data),digest)==(d['bytes'],d['sha256']);files_checked+=1;bytes_checked+=len(data)
    source={'artifact':p['artifact_id'],'view':c['label'],'path':str(f.relative_to(W)),'bytes':len(data),'sha256':digest}

     sid='regular:'+digest;seen[digest]=sid;result=scan(data,sid);result['source_id']=sid;candidates.append(result);zipscan(data,sid)
    source['scan_source_id']=seen[digest];local.append(source)
   sources.extend(local)
  for rel in p['remaining_encrypted_auxiliary']:

 hits=[dict(x,source_id=r['source_id'],content_sha256=r['content_sha256']) for r in candidates for x in r['candidate_hits']]
 plausible=[r for r in hits if r['status']=='PLAUSIBLE_SQLITE3_HEADER_REQUIRES_FULL_FORMAT_REVIEW']

 # Do not promote even a plausible header to actual recovered DB bytes automatically.
 report=f'''# F450/F459 embedded database format admission\n\nThis pass scans complete selected original regular-file payloads at every byte offset for SQLite3/SQLite2 and finite BDB/GDBM/LMDB signatures.
