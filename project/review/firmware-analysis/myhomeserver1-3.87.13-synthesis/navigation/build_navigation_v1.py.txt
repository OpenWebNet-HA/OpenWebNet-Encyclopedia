def sha(b): return hashlib.sha256(b).hexdigest()
def encode(x): return (json.dumps(x, sort_keys=True, indent=2, ensure_ascii=False)+'\n').encode()

def main():

    outputs,count=build()
    first=HERE/'first-candidate-v1'
    copies={**outputs,'build_navigation_v1.py':Path(__file__).read_bytes()}
    index=encode({name:{'bytes':len(data),'sha256':sha(data)} for name,data in sorted(copies.items())})
    if args.check:
        for name,data in outputs.items():assert (HERE/name).read_bytes()==data,name
        assert (first/'INDEX.json').read_bytes()==index,'snapshot index'
        assert sorted(p.name for p in first.iterdir())==sorted([*copies,'INDEX.json'])
        for name,data in copies.items():assert (first/name).read_bytes()==data,name
    else:
        assert not first.exists(),'refusing existing first-candidate directory'
        assert all(not (HERE/name).exists() for name in outputs),'refusing existing generated output'
        for name,data in outputs.items():
            with (HERE/name).open('xb') as f:f.write(data)
        first.mkdir()
        for name,data in copies.items():
            with (first/name).open('xb') as f:f.write(data)
        with (first/'INDEX.json').open('xb') as f:f.write(index)
    print(json.dumps({'mode':'check' if args.check else 'generate','outputs':len(outputs),'input_pins':count,
                      'rows':17,'completed_checkpoints':17,'protected_members':228,'saved_entries':8,
                      'saved_original_code_visits':340,'current_guest_execution':0,'current_guest_replay':0,
                      'first_copies':len(copies)},sort_keys=True))

if __name__=='__main__':main()
