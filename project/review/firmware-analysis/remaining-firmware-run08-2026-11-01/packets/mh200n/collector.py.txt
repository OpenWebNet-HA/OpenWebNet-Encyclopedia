def sha(b):
    return hashlib.sha256(b).hexdigest()


def enc(x):
    return (json.dumps(x, indent=2, sort_keys=True) + '\n').encode()


def exclusive(p, b):
    p.parent.mkdir(parents=True, exist_ok=True)
    with p.open('xb') as f:
        f.write(b)


def build():
    pins = {}

    def read(p):
        p = Path(p)
        f = ROOT / p
        assert not p.is_absolute() and '..' not in p.parts
        assert f.is_file() and not f.is_symlink() and ROOT in f.resolve().parents
        b = f.read_bytes()
        pins[str(p)] = {'path': str(p), 'bytes': len(b), 'sha256': sha(b)}
        return b

    manifest = json.loads(read(BASE / 'analysis/application-regular-files.json'))
    original = BASE / 'analysis/application/home/bticino/bin/bt_taskmanager'
    b = read(original)
    assert len(b) == manifest['home/bticino/bin/bt_taskmanager']['bytes'] == 159048
    assert sha(b) == manifest['home/bticino/bin/bt_taskmanager']['sha256'] == 'ad68f9bf43f7f10af122f34d3cb14b7d8c02730c46225a9a6cd1bf3e663072cb'
    assert b[:8] == bytes.fromhex('7f454c4601010161')
    h = struct.unpack_from('<HHIIIIIHHHHHH', b, 16)
    assert h[0:2] == (2, 40) and h[6] == 0x202 and h[8] == 32
    ph = [struct.unpack_from('<8I', b, h[4] + i * h[8]) for i in range(h[9])]
    loads = [{'offset': s[1], 'VA': s[2], 'file_bytes': s[4], 'memory_bytes': s[5], 'flags': s[6]} for s in ph if s[0] == 1]

    def off(a, n=4):
        xs = [s['offset'] + a - s['VA'] for s in loads if s['VA'] <= a and a + n <= s['VA'] + s['file_bytes']]
        assert len(xs) == 1, hex(a)
        return xs[0]

    def word(a):
        return struct.unpack_from('<I', b, off(a))[0]

    def branch(a, linked):
        w = word(a)
        assert w & 0x0f000000 == (0x0b000000 if linked else 0x0a000000)
        n = w & 0xffffff
        if n & (1 << 23):
            n -= 1 << 24
        return a + 8 + n * 4

    for a, v in GUARDS.items():
        assert word(a) == v, hex(a)
    calls = {0xa9d8: 0x9e54, 0xa9f0: 0x9f80, 0xaa30: 0x9d04}
    branches = {0xa998: 0xa9c8, 0xa9e4: 0xaa5c}
    for a, v in calls.items():
        assert branch(a, True) == v
    for a, v in branches.items():
        assert branch(a, False) == v
    joins = []
    for a, p, v in [(0xaa24, 0xaaa4, 0x2b918), (0xaa28, 0xaaa8, 0x2b928)]:
        w = word(a)
        assert w & 0x0f7f0000 == 0x051f0000 and a + 8 + (w & 4095) == p and word(p) == v
        joins.append({'instruction_VA': hex(a), 'pool_VA': hex(p), 'pool_word': hex(v)})
    literals = []
    for a, text in [(0x2b918, 'rm -f %s%s.zip'), (0x2b928, '/home/bticino/script/')]:
        raw = text.encode('ascii') + b'\0'
        assert b[off(a, len(raw)):off(a, len(raw)) + len(raw)] == raw
        literals.append({'VA': hex(a), 'file_offset': hex(off(a, len(raw))), 'text': text, 'NUL_bytes_hex': raw.hex(), 'role': 'Compiled formatter argument only, never a submitted command or live pathname'})
    rel = subprocess.check_output(['readelf', '-rW', str(ROOT / original)], text=True)

    def immediate(v):
        n = v & 255
        r = ((v >> 8) & 15) * 2
        return ((n >> r) | (n << (32 - r))) & 0xffffffff if r else n

    imports = []
    for name, a in [('fgets', 0x9e54), ('strlen', 0x9f80), ('sprintf', 0x9d04)]:
        u, v, z = [word(a + i) for i in (0, 4, 8)]
        assert u & 0xfffff000 == 0xe28fc000 and v & 0xfffff000 == 0xe28cc000 and z & 0xfffff000 == 0xe5bcf000
        slot = a + 8 + immediate(u) + immediate(v) + (z & 4095)
        rows = [x.strip() for x in rel.splitlines() if 'R_ARM_JUMP_SLOT' in x and re.search(r'\b' + name + r'@GLIBC_', x)]
        assert len(rows) == 1 and int(rows[0].split()[0], 16) == slot
        imports.append({'name': name, 'PLT_VA': hex(a), 'GOT_slot': hex(slot), 'words': [hex(u), hex(v), hex(z)], 'original_relocation': rows[0]})
    spans = []
    for label, a, z in SPANS:

        raw = b[o:o + z - a]

        spans.append({'label': label, 'start_VA': hex(a), 'end_exclusive_VA': hex(z), 'file_offset': hex(o), 'bytes_hex': raw.hex(), 'sha256': sha(raw), 'host_disassembly': asm})
    prior = BASE / 'research/2026-11-01-night07-application'

              'store_address_algebra': {'line_buffer': 'FP-116', 'L': 'ordinary strlen return for that line', 'MVN_R2_104_signed': -105, 'FP_base': 'FP-12', 'selected_STRB_address': '(FP-12)+L-105=(FP-116)+L-1', 'stored_byte': 0, 'premise': 'Valid nonempty terminated line and writable last-byte address; not a newline test'},

    out = build()
    if a.check:
        for n, b in out.items():
            assert (HERE / n).read_bytes() == b, n
        for x in json.loads((HERE / 'first-candidate/INDEX.json').read_bytes())['files']:
